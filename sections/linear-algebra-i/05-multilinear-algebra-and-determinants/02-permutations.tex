\subsection{Permutations and composition}

Antisymmetry concerns the order of the inputs to an area or volume
function. Permutations provide a way to describe every such reordering.
Their signs will record the corresponding changes of orientation in
the determinant formula.

\begin{definition}
  A permutation of \(\{1,\ldots,n\}\) is a bijective map
  % Author-requested combinatorial context, October 6, 2026.
  \marginnote{%
    \textbf{Permutations in combinatorics.}
    A permutation is an ordered arrangement of distinct objects, without
    repetition. The list \((\sigma(1),\ldots,\sigma(n))\) identifies a full
    arrangement with a bijection in \(S_n\). There are \(n\) choices for the
    first entry, \(n-1\) for the next, and so on, giving \(n!\) arrangements.
    \par
    For \(0\le k\le n\), the binomial coefficient
    \[
      \binom{n}{k}=\frac{n!}{k!(n-k)!}
    \]
    counts unordered \(k\)-element subsets. Each subset has \(k!\) orderings,
    so the number of ordered selections of \(k\) distinct objects is
    \[
      k!\binom{n}{k}=\frac{n!}{(n-k)!},
    \]
    with \(0!=1\). These same binomial coefficients appear in \((x+y)^n\).
    Choosing \(k\) of the \(n\) factors to contribute \(y\) gives the term
    \(\binom{n}{k}x^{n-k}y^k\).
  }
  \[
    \sigma:\{1,\ldots,n\}\longrightarrow\{1,\ldots,n\}.
  \]
  Write \(S_n\) for the set of all such permutations.
\end{definition}

Permutations reorder \(n\) objects. Under composition, \(S_n\) is a group.
The identity permutation \(\iota\) belongs to \(S_n\), and the composition
of two permutations is again a permutation. Every \(\sigma\in S_n\) has
an inverse satisfying
\[
  \sigma\circ\sigma^{-1}=\sigma^{-1}\circ\sigma=\iota.
\]
Composition is associative.

\begin{definition}
  A transposition switches two elements and leaves all the others unchanged.
  For \(i\ne j\), the transposition \(\tau_{ij}\in S_n\) is
  \[
    \tau_{ij}(k)=
    \begin{cases}
      j & \text{if \(k=i\)},\\
      i & \text{if \(k=j\)},\\
      k & \text{otherwise}.
    \end{cases}
  \]
\end{definition}

\begin{example}
  The transposition \(\tau_{24}\in S_4\) satisfies
  \[
    \tau_{24}(1)=1,\quad \tau_{24}(2)=4,\quad
    \tau_{24}(3)=3,\quad \tau_{24}(4)=2.
  \]
\end{example}

\begin{lemma}[Decomposition into transpositions]
  \label{thm:linear-algebra-i:transposition-decomposition}
  Every permutation can be written as a composition of transpositions,
  \[
    \sigma=\tau_k\circ\cdots\circ\tau_1.
  \]
  The identity is the composition of zero transpositions.
\end{lemma}
% PDF pp. 20--21 and IMG_9209 state decomposition without a proof.
% The following proof was authorized on November 10, 2026.
\begin{pendingreview}
\begin{proof}
  The identity already has the required empty decomposition. For a
  permutation \(\sigma\) that is not the identity, choose \(r\) with
  \(\sigma(r)\ne r\), and put \(s=\sigma(r)\). Compose on the left
  with \(\tau_{rs}\). The new permutation fixes \(r\), because
  \[
    (\tau_{rs}\circ\sigma)(r)=\tau_{rs}(s)=r.
  \]
  It also preserves every index previously fixed by \(\sigma\).
  Such an index cannot be \(r\), and it cannot be \(s\), since
  \(\sigma(s)=s=\sigma(r)\) would contradict injectivity.
  The transposition therefore leaves each previously fixed value unchanged.

  Repeating this operation increases the number of fixed indices at each
  step, so after at most \(n\) steps it produces the identity.
  Every transposition is its own inverse. Undoing these successive
  compositions consequently expresses the original \(\sigma\) as a
  composition of transpositions.
\end{proof}
\end{pendingreview}

\begin{example}
  The permutation \(\sigma\in S_3\) given by
  \[
    \sigma(1)=2,\qquad \sigma(2)=3,\qquad \sigma(3)=1
  \]
  is \(\sigma=\tau_{13}\circ\tau_{12}\). Here
  \[
    \begin{aligned}
      \tau_{12}(1)&=2,& \tau_{12}(2)&=1,& \tau_{12}(3)&=3,\\
      \tau_{13}(1)&=3,& \tau_{13}(2)&=2,& \tau_{13}(3)&=1.
    \end{aligned}
  \]
  The paths in \autoref{fig:linear-algebra-i:permutation-composition}
  track each element through these two transpositions.

\end{example}

\subsection{Parity and the sign function}

\begin{proposition}[Parity of a permutation]
  \label{thm:linear-algebra-i:permutation-parity}
  A permutation cannot be both a composition of an even number of
  transpositions and a composition of an odd number of transpositions.
\end{proposition}
% PDF p. 23 and IMG_9209 state parity without its proof. The following
% determinant-free product argument was authorized on November 10, 2026.
\begin{pendingreview}
\begin{proof}
  For real numbers \(a_1,\ldots,a_n\), define
  \[
    \Delta(a_1,\ldots,a_n)=\prod_{1\leq i<j\leq n}(a_j-a_i).
  \]
  An empty product is \(1\). We first show that exchanging any two
  entries changes the sign of \(\Delta\). Suppose their positions are
  \(r<s\). The factor \(a_s-a_r\) changes sign. For each index
  outside the interval from \(r\) to \(s\), the two factors involving
  that index and \(r,s\) simply exchange places, leaving their product
  unchanged. For \(r<j<s\), the two affected factors have product
  \[
    (a_j-a_r)(a_s-a_j).
  \]
  After the exchange, their product is
  \[
    (a_j-a_s)(a_r-a_j)=(a_s-a_j)(a_j-a_r),
  \]
  which is the same scalar. Every other factor is unchanged. Only the
  factor belonging to \(r,s\) contributes a minus sign, proving the claim.

  Now start with the list \((1,\ldots,n)\) and build \(\sigma\) by
  composing transpositions on the left. At each stage, a transposition
  interchanges the two occurrences of its moved values in the current
  permutation list. The preceding calculation therefore shows that a
  decomposition with \(k\) transpositions gives
  \[
    \Delta(\sigma(1),\ldots,\sigma(n))
      =(-1)^k\Delta(1,\ldots,n).
  \]
  The number \(\Delta(1,\ldots,n)\) is nonzero, since every factor
  \(j-i\) is positive. Thus \((-1)^k\) is determined by \(\sigma\)
  itself. Any second decomposition must have the same parity, so an even
  decomposition and an odd decomposition cannot represent the same permutation.
\end{proof}
\end{pendingreview}

Consequently, the following definition does not depend on the chosen
transposition decomposition. Different decompositions may have different
lengths, but those lengths have the same parity.

\begin{definition}
  The sign, or parity, of a permutation is the function
  \(\varepsilon:S_n\to\{-1,1\}\). For a decomposition
  \(\sigma=\tau_k\circ\cdots\circ\tau_1\) into transpositions,
  \[
    \varepsilon(\sigma)=
    \begin{cases}
      1 & \text{if \(k\) is even},\\
      -1 & \text{if \(k\) is odd}.
    \end{cases}
  \]
\end{definition}

The sign function has the properties
\[
  \begin{aligned}
    \varepsilon(\iota)&=1,\\
    \varepsilon(\tau)&=-1 &&\text{for every transposition \(\tau\)},\\
    \varepsilon(\sigma_2\circ\sigma_1)
      &=\varepsilon(\sigma_2)\varepsilon(\sigma_1),\\
    \varepsilon(\sigma^{-1})&=\varepsilon(\sigma).
  \end{aligned}
\]
% Verification of the first three identities was authorized November 10, 2026.
\begin{pendingreview}
\begin{proof}[Verification of the first three identities]
  The identity has a decomposition of length zero, so its sign is \(1\).
  A transposition has a decomposition of length one, so its sign is \(-1\).
  If \(\sigma_1\) is a composition of \(p\) transpositions and
  \(\sigma_2\) is a composition of \(q\) transpositions, concatenating
  the two decompositions expresses \(\sigma_2\circ\sigma_1\) as a
  composition of \(p+q\) transpositions. The definition of sign gives
  \[
    \varepsilon(\sigma_2\circ\sigma_1)
      =(-1)^{p+q}=(-1)^q(-1)^p
      =\varepsilon(\sigma_2)\varepsilon(\sigma_1).
  \]
  This proves the identity, transposition, and composition formulas.
\end{proof}
\end{pendingreview}

For the last property, reversing the order of the same transpositions gives
the inverse,
\[
  \sigma=\tau_1\circ\cdots\circ\tau_N
  \quad\Longrightarrow\quad
  \sigma^{-1}=\tau_N\circ\cdots\circ\tau_1.
\]
The sign function is uniquely determined by assigning sign \(-1\) to
every transposition and requiring multiplicativity under composition.
Its existence depends on \autoref{thm:linear-algebra-i:permutation-parity}.

\subsection{Extending the sign to all maps}

Let \(T_n\) be the set of all maps
\[
  \phi:\{1,\ldots,n\}\longrightarrow\{1,\ldots,n\}.
\]
Then \(S_n\subseteq T_n\). Extend the sign function to
\(\varepsilon:T_n\to\{-1,0,1\}\) by keeping its existing values on
\(S_n\) and setting
\[
  \varepsilon(\phi)=0\qquad\text{if \(\phi\notin S_n\)}.
\]
