\begin{exercise}
  \label{ex:hw07:coordinate-inner-products}
  Which of the following is an inner product? Explain.
  % Protect the literal question prefix; measure the last subpart label.
  \begin{enumerate}[{1}.1.][2]
    \item (5 points) For any \((v^1,v^2),(w^1,w^2)\in\R^2\),
      \[
        (v,w)=v^1w^2+v^2w^1.
      \]
    \item (5 points) For any \((v^1,v^2),(w^1,w^2)\in\R^2\),
      \[
        (v,w)=2v^1w^1-v^1w^2-v^2w^1+2v^2w^2.
      \]
  \end{enumerate}

\end{exercise}
