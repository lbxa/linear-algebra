\begin{exercise}
  \label{ex:hw07:tensor-inner-products}
  Let \(V\) be a 2-dimensional real vector space and \((e_1,e_2)\)
  be a basis. Which of the following 2-tensors is an inner product?
  Explain.
  \begin{enumerate}[{2}.1.][2]
    \item (5 points) Let \(T:V\times V\to\R\) be the 2-tensor given by
      \[
        \begin{aligned}
          T(e_1,e_1)&=1\\
          T(e_1,e_2)=T(e_2,e_1)&=1\\
          T(e_2,e_2)&=1
        \end{aligned}
      \]
    \item (5 points) Let \(T:V\times V\to\R\) be given by
      \[
        \begin{aligned}
          T(e_1,e_1)&=1\\
          T(e_1,e_2)&=1\\
          T(e_2,e_1)&=-1\\
          T(e_2,e_2)&=1
        \end{aligned}
      \]
  \end{enumerate}
  \solutionlink{hw07:tensor-inner-products}
\end{exercise}
