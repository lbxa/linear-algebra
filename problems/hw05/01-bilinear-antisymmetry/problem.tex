\begin{exercise}
  \label{ex:hw05:bilinear-antisymmetry}
  \solutionlink{hw05:bilinear-antisymmetry}
  Let \(A:V\times V\to\R\) be a bilinear function on a real vector
  space \(V\) such that \(A(v,v)=0\) for every \(v\in V\).
  Show \(A\) is antisymmetric.
\end{exercise}
