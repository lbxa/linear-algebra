\begin{exercise}
  \label{ex:hw05:column-addition}
  \solutionlink{hw05:column-addition}
  Let the columns of \(M\) be \(c_1,\ldots,c_n\). Prove that, given
  \(j\ne k\) and \(a\in\mathbb F\), if \(M'\) is the matrix whose
  columns are given by
  \[
    \begin{aligned}
      c'_j&=c_j+ac_k\\
      c'_i&=c_i\quad\text{if \(i\ne j\)}
    \end{aligned}
  \]
  then
  \[
    \det(M')=\det(M).
  \]
\end{exercise}
