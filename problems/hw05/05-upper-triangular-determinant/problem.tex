\begin{exercise}
  \label{ex:hw05:upper-triangular-determinant}
  \solutionlink{hw05:upper-triangular-determinant}
  Explain why the determinant of an upper triangular \(m\times m\)
  matrix equals the product of its diagonal entries. In other words,
  explain why
  \[
    \det\begin{bmatrix}
      A^1_1&A^1_2&\cdots&A^1_{m-1}&A^1_m\\
      0&A^2_2&\cdots&A^2_{m-1}&A^2_m\\
      \vdots&&&&\vdots\\
      0&0&\cdots&0&A^m_m
    \end{bmatrix}=A^1_1 A^2_2\cdots A^m_m.
  \]
  A detailed proof is not required; describe the process that gives
  this conclusion.
\end{exercise}
