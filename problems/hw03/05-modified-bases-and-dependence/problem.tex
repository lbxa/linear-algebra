\begin{exercise}
  \label{ex:hw03:modified-bases-and-dependence}
  Let \(\{u,v,w\}\) be a basis of a real vector space \(V\).
  \begin{enumerate}[(a)][2]
    \item Show that \(\{u+v+w,v+w,w\}\) is also a basis.
    \item Is it possible that a subset \(\{v_1,v_2,v_3\}\subset V\)
      is linearly dependent but the set
      \(\{v_2+v_3,v_3+v_1,v_1+v_2\}\) is linearly independent?
  \end{enumerate}

\end{exercise}
