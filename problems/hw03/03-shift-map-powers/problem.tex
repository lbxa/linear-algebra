\begin{exercise}
  \label{ex:hw03:shift-map-powers}
  Let \(E=\begin{bmatrix}e_1&e_2&e_3\end{bmatrix}\) be a basis of
  \(V\). Consider the linear map \(L:V\to V\) such that
  \[
    \begin{aligned}
      L(e_1)&=e_2\\
      L(e_2)&=e_3\\
      L(e_3)&=0
    \end{aligned}
  \]
  Show that if \(M\) is the matrix where \(L(E)=EM\), then
  \(M,M^2\ne0\) but \(M^3=0\).
  \solutionlink{hw03:shift-map-powers}
\end{exercise}
