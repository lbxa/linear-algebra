\begin{exercise}
  \label{ex:hw02:one-sided-inverses}
  Let \(A,B\in\mathcal M_{m\times m}\) with \(AB=I\).
  \begin{enumerate}[(a)][2]
    \item Prove that if \(x\) is a column matrix such that \(Bx=0\),
      then \(x=0\).
    \item Deduce that the columns of \(B\) are distinct and form a
      linearly independent subset of \(\mathbb F^m\) with \(m\)
      elements, and hence a basis. Conclude that every \(y\in\mathbb F^m\)
      can be written as \(y=Bx\).
    \item Prove that \(BA=I\).
  \end{enumerate}
  \solutionlink{hw02:one-sided-inverses}
\end{exercise}
