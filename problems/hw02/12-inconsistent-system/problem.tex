\begin{exercise}
  \label{ex:hw02:inconsistent-system}
  Work over \(\mathbb F=\R\) or \(\C\). Consider the system
  \[
    2x-3y=5,\qquad x+y=-1,\qquad 3x+5y=0,
  \]
  \begin{enumerate}[(a)][2]
    \item Show that this system has no solution.
    \item Find matrices \(A,v,\) and \(b\) such that the system is
      equivalent to
      \[
        Av=b.
      \]
    \item Find the set of all \(b\in\mathbb F^3\) for which \(Av=b\)
      has a solution and show that given such a \(b\), the solution
      is unique
  \end{enumerate}
  \solutionlink{hw02:inconsistent-system}
\end{exercise}
