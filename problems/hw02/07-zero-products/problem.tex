\begin{exercise}
  \label{ex:hw02:zero-products}
  There are nonzero matrices \(A\) and \(B\) such that \(AB=0\).
  \begin{enumerate}[(a)][2]
    \item Find all \(A\in\mathcal M_{2\times2}\) with \(A^2=0\).
    \item Find \(A,B\in\mathcal M_{2\times2}\) with \(AB=0\) and
      \(BA\ne0\).
  \end{enumerate}

\end{exercise}
