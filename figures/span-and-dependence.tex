% Two-vector configurations in R^2; dependence is not generally collinearity
% for lists of more than two vectors.
\begin{tikzpicture}[
  x=10mm,y=10mm,font=\footnotesize,line width=0.45pt,
  vector/.style={-{Latex[length=1.6mm,width=1.1mm]}},
  axis/.style={vector,black!40,line width=0.35pt},
  guide/.style={black!20,line width=0.3pt}
]
  \begin{scope}
    \node at (1.2,2.8) {Independent pair};
    \draw[axis] (-0.5,0) -- (2.9,0);
    \draw[axis] (0,-0.4) -- (0,2.4);
    \begin{scope}
      \clip (-0.6,-0.7) rectangle (2.9,2.4);
      \foreach \a in {-2,-1,0,1,2,3} {
        \draw[guide] ({1.6*\a-1.6},{0.3*\a-3.2}) -- ({1.6*\a+1.6},{0.3*\a+3.2});
        \draw[guide] ({0.8*\a-3.2},{1.6*\a-0.6}) -- ({0.8*\a+3.2},{1.6*\a+0.6});
      }
    \end{scope}
    \draw[vector] (0,0) -- (1.6,0.3) node[right=2pt] {\(v\)};
    \draw[vector] (0,0) -- (0.8,1.6) node[above left=2pt] {\(w\)};
    \node[below left=2pt] at (0,0) {\(0\)};
    \node at (1.2,-1.2) {\(\operatorname{span}\{v,w\}=\R^2\)};
  \end{scope}
  \begin{scope}[xshift=58mm]
    \node at (1.2,2.8) {Dependent pair};
    \draw[axis] (-0.5,0) -- (2.9,0);
    \draw[axis] (0,-0.4) -- (0,2.4);
    \draw[black!45,{Latex[length=1.6mm,width=1.1mm]}-{Latex[length=1.6mm,width=1.1mm]}]
      (-0.6,-0.4) -- (2.8,{2.8*2/3});
    \draw[vector] (0,0) -- (2.1,1.4) node[above=3pt] {\(w\)};
    \fill (1.05,0.7) circle[radius=1.3pt];
    \node[below right=2pt] at (1.05,0.7) {\(v\)};
    \node[below left=2pt] at (0,0) {\(0\)};
    \node at (1.2,-1.2) {\(\operatorname{span}\{v,w\}=\operatorname{span}\{v\}\)};
  \end{scope}
\end{tikzpicture}
